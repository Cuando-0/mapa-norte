function doPost(e) { try { // 1. Recibir y parsear el comando enviado desde el HTML var datos = JSON.parse(e.postData.contents); var comando = datos.comando.trim(); var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // 2. Expresión regular para entender el comando // Soporta: "PLACA1 [comando] PLACA2" var regex = /^([A-Z0-9]+)\s+(enfrente de|atras de|a la derecha de|a la izquierda de)\s+([A-Z0-9]+)$/i; var match = comando.match(regex); if (!match) { return ContentService.createTextOutput("Comando con formato incorrecto."); } var placaMover = match[1].toUpperCase(); var accion = match[2].toLowerCase(); var placaReferencia = match[3].toUpperCase(); // 3. Buscar la celda de la placa de referencia en el mapa var textFinder = sheet.createTextFinder(placaReferencia); var celdaReferencia = textFinder.findNext(); if (!celdaReferencia) { return ContentService.createTextOutput("No se encontró la placa de referencia: " + placaReferencia); } var filaRef = celdaReferencia.getRow(); var colRef = celdaReferencia.getColumn(); // 4. Determinar la nueva celda según el comando de posición var nuevaFila = filaRef; var nuevaCol = colRef; switch (accion) { case "enfrente de": // Mirando el mapa de tu imagen, "enfrente" suele ser la fila de arriba (hacia la carretera 1) nuevaFila = filaRef - 1; break; case "atras de": // "Atrás" suele ser la fila de abajo (hacia la carretera 2 o rampas) nuevaFila = filaRef + 1; break; case "a la derecha de": nuevaCol = colRef + 1; break; case "a la izquierda de": nuevaCol = colRef - 1; break; } // 5. Buscar si la placa ya existía en el mapa para borrarla de su posición vieja var buscadorViejo = sheet.createTextFinder(placaMover); var celdaVieja = buscadorViejo.findNext(); if (celdaVieja) { celdaVieja.setValue(""); celdaVieja.setBackground("#FFFFFF"); // O el color gris de tu fondo si prefieres } // 6. Asignar la placa a la nueva celda y pintarla de azul claro var nuevaCelda = sheet.getRange(nuevaFila, nuevaCol); nuevaCelda.setValue(placaMover); nuevaCelda.setBackground("#9cc2e5"); // <--- Aquí se aplica tu color solicitado return ContentService.createTextOutput("Posición actualizada con éxito."); } catch(error) { return ContentService.createTextOutput("Error en el script: " + error.toString()); } }
